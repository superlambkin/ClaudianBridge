## 🔄 データ移行（P2 初回起動時）

```typescript
async function migrateFromLegacy(store: ConfigStore): Promise<{ migrated: string[] }> {
  const result = { migrated: [] };
  const csb = readDataJsonSafe('.obsidian/plugins/claudian-selection-bridge/data.json');
  if (csb && !store.load().general.migratedFrom.claudianSelectionBridge) {
    store.save({ ...store.load(), selection: { enabled: csb.enabled ?? true, delayMs: csb.delayMs ?? 300 }, tts: { enabled: true, engine: csb.tts.engine, voices: csb.tts.voices, minimax: csb.tts.minimax, voice: csb.tts.voice }, general: { ...store.load().general, migratedFrom: { ...store.load().general.migratedFrom, claudianSelectionBridge: true } } });
    backup('.obsidian/plugins/claudian-selection-bridge/data.json');
    result.migrated.push('claudian-selection-bridge');
  }
  return result;
}
```

## 🚫 旧プラグイン無効化（P2 初回起動時、1度だけ実行）

```typescript
async function disableLegacyPluginsOnce(): Promise<void> {
  const flag = '.obsidian/.claudian-bridge.legacy-disabled';
  if (await exists(flag)) return;

  // 1. community-plugins.json から3プラグイン除外
  const cpPath = '.obsidian/community-plugins.json';
  const cp = JSON.parse(await readFile(cpPath, 'utf-8'));
  const remove = ['claudian-selection-bridge', 'vault-office-bridge', 'extension-whitelist'];
  cp.plugins = cp.plugins.filter((p: string) => !remove.includes(p));
  await writeFile(cpPath, JSON.stringify(cp, null, 2));

  // 2. フォルダを _disabled__ 接頭辞でリネーム
  for (const id of ['claudian-selection-bridge', 'vault-office-bridge']) {
    await rename(`.obsidian/plugins/${id}`, `.obsidian/plugins/_disabled__${id}`).catch(() => {});
  }

  await writeFile(flag, new Date().toISOString());
}
```

## 📝 TTS 実装の引き継ぎ

`claudettsHttpSpeak` の最新完成版（stdin UTF-8 + settled guard 込み）を正確に移植：

```typescript
async function claudettsHttpSpeak(text: string, settings: TtsSettings, noticeFn: (m: string) => void): Promise<boolean> {
  return new Promise((resolve) => {
    let settled = false;
    const cmd = path.join(os.homedir(), '.claude', 'skills', 'claude-tts', 'scripts', 'commands.py');
    let child: ChildProcess;
    try {
      child = spawn('python', [cmd, 'speak'], { windowsHide: true });
    } catch (e: any) {
      noticeFn(`⚠️ ClaudeTTS 起動失敗: ${e.message}`);
      resolve(false);
      return;
    }
    let err = '';
    child.stderr?.on('data', (d: Buffer) => (err += d.toString()));
    child.on('error', (e) => {
      if (settled) return;
      settled = true;
      noticeFn(`⚠️ ClaudeTTS 失敗: ${e.message}`);
      resolve(false);
    });
    child.on('close', (code) => {
      if (settled) return;
      settled = true;
      if (code === 0) resolve(true);
      else {
        noticeFn(`⚠️ ClaudeTTS 失敗 (exit ${code}): ${err.slice(0, 200)}`);
        resolve(false);
      }
    });
    child.stdin?.write(text);
    child.stdin?.end();
  });
}
```
